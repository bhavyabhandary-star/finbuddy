Build a complete, fully interactive single-page web app called FinBuddy. It is an AI-powered credit assessment tool for micro-entrepreneurs and gig workers in India. All state is managed in JavaScript — no page reloads. When the user taps the CTA button, the app makes a POST request to a webhook that returns a JSON credit assessment directly. The response is rendered on Screen 2.

================================================================
API INTEGRATION
================================================================

const WEBHOOK_URL = "WEBHOOK_URL";

// The n8n workflow returns FinBuddy JSON directly — no unwrapping needed.
function runAssessment() {
  const payload = {
    persona_id: selectedPersona,   // "rajesh" | "meena" | "custom"
    language:   selectedLanguage   // "hindi" | "english" | "tamil" | "telugu"
  };

  fetch(WEBHOOK_URL, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload)
  })
  .then(r => { if (!r.ok) throw new Error("HTTP " + r.status); return r.json(); })
  .then(data => renderResults(data))
  .catch(() => {
    renderResults(DEMO_DATA);
    showToast("Webhook unreachable — showing demo data.", "warning");
  });
}

// renderResults reads these fields directly from the response:
// data.decision, data.approved_amount, data.confidence,
// data.shap_factors, data.roadmap, data.vernacular_message, data.minimum_offer
